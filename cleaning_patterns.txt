# Regex patterns to remove from the END of organization names
# These are metadata about bill relationships, not part of the org name
# Each pattern starts with (?i) to make it case-insensitive
# Patterns support both round brackets () and square brackets []

# Round brackets - END
(?i)\s*\(co-sponsor\)\s*$
(?i)\s*\(sponsor\)\s*$
(?i)\s*\(sponsors\)\s*$
(?i)\s*\(co-source\)\s*$
(?i)\s*\(prior version\)\s*$
(?i)\s*\(oppose\)\s*$
(?i)\s*\(oppose unless amended\)\s*$
(?i)\s*\("in concept"\)\s*$
(?i)\s*\(principal co-sponsor\)\s*$
(?i)\s*\(co-sponsors\)\s*$
(?i)\s*\(previous version\)\s*$
(?i)\s*\(as of prior version\)\s*$
(?i)\s*\(unable to re-verify at time of writing\)\s*$
(?i)\s*\(none received[^)]*\)\s*$
(?i)\s*\(reflects[^)]*\)\s*$
(?i)\s*\([^)]*\sversion[^)]*\)\s*$
(?i)\s*\(prior to latest amendments\)\s*$
(?i)\s*\(per\s[^)]*\)\s*$
(?i)\s*\(as of\s[^)]*\)\s*$

# General: trailing round brackets containing metadata keywords - END
(?i)\s*\([^)]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^)]*\)\s*$

# General: trailing square brackets containing metadata keywords - END
(?i)\s*\[[^\]]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^\]]*\]\s*$

# Trailing bare date stamps with year component - END
\s+-\s*\d{1,2}/\d{1,2}/\d{2,4}\s*$
(?<=\D)\s+\d{1,2}/\d{1,2}/\d{2,4}\s*$

# Trailing "None Received" - END
(?i)\s+None Received\s*$

# Trailing bare position words - END
(?i),?\s+oppose\s*$
(?i)\s+OPPOSED\s*$

# Square brackets - END
(?i)\s*\[co-sponsor\]\s*$
(?i)\s*\[sponsor\]\s*$
(?i)\s*\[sponsors\]\s*$
(?i)\s*\[co-source\]\s*$
(?i)\s*\[prior version\]\s*$
(?i)\s*\[oppose\]\s*$
(?i)\s*\[oppose unless amended\]\s*$
(?i)\s*\["in concept"\]\s*$
(?i)\s*\[principal co-sponsor\]\s*$
(?i)\s*\[co-sponsors\]\s*$
(?i)\s*\[previous version\]\s*$
(?i)\s*\[as of prior version\]\s*$
(?i)\s*\[unable to re-verify at time of writing\]\s*$
(?i)\s*\[none received[^\]]*\]\s*$
(?i)\s*\[reflects[^\]]*\]\s*$
(?i)\s*\[[^\]]*\sversion[^\]]*\]\s*$
(?i)\s*\[prior to latest amendments\]\s*$
(?i)\s*\[per\s[^\]]*\]\s*$
(?i)\s*\[as of\s[^\]]*\]\s*$
