# Regex patterns to remove from the END of organization names
# These are metadata about bill relationships, not part of the org name
# Each pattern starts with (?i) to make it case-insensitive
# Patterns support both round brackets () and square brackets []

# Round brackets - END
(?i)\s*\(co-sponsor\)\s*$
(?i)\s*\(sponsor\)\s*$
(?i)\s*\(sponsors\)\s*$
(?i)\s*\(co-source\)\s*$
(?i)\s*\(prior version\)\s*$
(?i)\s*\(oppose\)\s*$
(?i)\s*\(oppose unless amended\)\s*$
(?i)\s*\("in concept"\)\s*$
(?i)\s*\(principal co-sponsor\)\s*$
(?i)\s*\(co-sponsors\)\s*$
(?i)\s*\(previous version\)\s*$
(?i)\s*\(as of prior version\)\s*$
(?i)\s*\(unable to re-verify at time of writing\)\s*$
(?i)\s*\(none received[^)]*\)\s*$
(?i)\s*\(reflects[^)]*\)\s*$
(?i)\s*\([^)]*\sversion[^)]*\)\s*$
(?i)\s*\(prior to latest amendments\)\s*$
(?i)\s*\(per\s[^)]*\)\s*$
(?i)\s*\(as of\s[^)]*\)\s*$

# General: trailing round brackets containing metadata keywords - END
(?i)\s*\([^)]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^)]*\)\s*$

# General: trailing square brackets containing metadata keywords - END
(?i)\s*\[[^\]]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^\]]*\]\s*$

# Trailing bare date stamps with year component - END
\s+-\s*\d{1,2}/\d{1,2}/\d{2,4}\s*$
(?<=\D)\s+\d{1,2}/\d{1,2}/\d{2,4}\s*$

# Trailing "None Received" - END
(?i)\s+None Received\s*$

# Trailing bare position words - END
(?i),?\s+oppose\s*$
(?i)\s+OPPOSED\s*$
(?i),?\s+Continued\s*$

# Trailing "None on file" suffix - END
(?i),?\s*None on file\s*$

# Square brackets - END
(?i)\s*\[co-sponsor\]\s*$
(?i)\s*\[sponsor\]\s*$
(?i)\s*\[sponsors\]\s*$
(?i)\s*\[co-source\]\s*$
(?i)\s*\[prior version\]\s*$
(?i)\s*\[oppose\]\s*$
(?i)\s*\[oppose unless amended\]\s*$
(?i)\s*\["in concept"\]\s*$
(?i)\s*\[principal co-sponsor\]\s*$
(?i)\s*\[co-sponsors\]\s*$
(?i)\s*\[previous version\]\s*$
(?i)\s*\[as of prior version\]\s*$
(?i)\s*\[unable to re-verify at time of writing\]\s*$
(?i)\s*\[none received[^\]]*\]\s*$
(?i)\s*\[reflects[^\]]*\]\s*$
(?i)\s*\[[^\]]*\sversion[^\]]*\]\s*$
(?i)\s*\[prior to latest amendments\]\s*$
(?i)\s*\[per\s[^\]]*\]\s*$
(?i)\s*\[as of\s[^\]]*\]\s*$

# Trailing "(if)", "(as)", "(c osponsor )", "(Recommendation)", "(page)", "(Member)" metadata - END
(?i)\s*\(if\)\s*$
(?i)\s*\(as\)\s*$
(?i)\s*\(c\s*osponsor\s*\)\s*$
(?i)\s*\(recommendation\)\s*$
(?i)\s*\(page\)\s*$
(?i)\s*\(member\)\s*$

# Trailing "(N letters)" or "(N individuals)" metadata suffix - END
(?i)\s*\(\d+\s+(?:letters|individuals)\)\s*$
(?i)\s*-\s*\(\d+\s+(?:letters|individuals)\)\s*$

# Trailing "(Month Day" date fragment (e.g. "(July 10", "(April 23") - END
(?i)\s*\((?:January|February|March|April|May|June|July|August|September|October|November|December)\s+\d+\s*$

# These are procedural notes, not part of the org name

(?i)^\s*\(prior to latest amendments\)\s*
(?i)^\s*\(per\s[^)]*\)\s*

# Leading "COMMENTS :" / "ARGUMENTS IN SUPPORT:" / "Staff Comments:" - BEGINNING
# NOTE: These must come BEFORE "According to" patterns so both prefixes get stripped
(?i)^\s*COMMENTS\s*:\s*
(?i)^\s*ARGUMENTS IN SUPPORT\s*:?\s*
(?i)^\s*Staff Comments\s*:\s*

# Leading "Support." / "SUPPORT." / "Support:" / "Support and Opposition." prefix - BEGINNING
(?i)^\s*Support and Opposition[\.:]\s*
(?i)^\s*SUPPORT[\.:]\s*

(?i)^\s*According to the\s+
(?i)^\s*According to\s+

# Leading "year: " prefix (e.g. "1998: American Farmland Trust") - BEGINNING
^\d{4}:\s+

# Leading "year) " prefix (e.g. "2007) American Cancer Society (July 10") - BEGINNING
^\d{4}\)\s+

# Leading "None on file" prefix - BEGINNING
(?i)^\s*None on file\.\s*
(?i)^\s*None on file\s+

# Leading "and " fragment - BEGINNING
(?i)^and\s+

# Leading numbered-list artifact (e.g. "1. Org Name", "1995. Org Name") - BEGINNING
(?i)^\s*\d+\.\s+

# General: leading round brackets containing metadata keywords - BEGINNING
(?i)^\s*\([^)]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^)]*\)\s*

# General: leading square brackets containing metadata keywords - BEGINNING
(?i)^\s*\[[^\]]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d)[^\]]*\]\s*

# Square brackets - BEGINNING
(?i)^\s*\[co-sponsor\]\s*
(?i)^\s*\[sponsor\]\s*
(?i)^\s*\[sponsors\]\s*
(?i)^\s*\[co-source\]\s*

# Patterns to remove from ANYWHERE in organization names
# These phrases can appear in the middle of strings and should be removed

# [pic] image placeholder artifacts - ANYWHERE
(?i)\[pic\]

# Round brackets - ANYWHERE
(?i)\s*\(submitted on earlier version\)\s*
(?i)\s*\(submitted on previous version\)\s*
(?i)\s*\(unable to re-verify\)\s*

# Square brackets - ANYWHERE
