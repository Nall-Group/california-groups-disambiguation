# Regex patterns to remove from the END of organization names
# These are metadata about bill relationships, not part of the org name
# Each pattern starts with (?i) to make it case-insensitive
# Patterns support both round brackets () and square brackets []

# Round brackets - END
(?i)\s*\(co-sponsor\)\s*$
(?i)\s*\(sponsor\)\s*$
(?i)\s*\(sponsors\)\s*$
(?i)\s*\(co-source\)\s*$
(?i)\s*\(prior version\)\s*$
(?i)\s*\(oppose\)\s*$
(?i)\s*\(oppose unless amended\)\s*$
(?i)\s*\("in concept"\)\s*$
(?i)\s*\(principal co-sponsor\)\s*$
(?i)\s*\(co-sponsors\)\s*$
(?i)\s*\(previous version\)\s*$
(?i)\s*\(as of prior version\)\s*$
(?i)\s*\(unable to re-verify at time of writing\)\s*$
(?i)\s*\(none received[^)]*\)\s*$
(?i)\s*\(reflects[^)]*\)\s*$
(?i)\s*\([^)]*\sversion[^)]*\)\s*$
(?i)\s*\(prior to latest amendments\)\s*$
(?i)\s*\(per\s[^)]*\)\s*$
(?i)\s*\(as of\s[^)]*\)\s*$

# Trailing "(updated date)" / "(dated date)" metadata - END
(?i)\s*\(updated\s+[^)]+\)\s*$
(?i)\s*\(dated\s+[^)]+\)\s*$

# Trailing "/prior version" without parens - END
(?i)\s*/prior version\s*$

# Trailing "- prior version" bare dash (no parens) - END
(?i)\s*[-–—]\s*prior version\s*$

# General: trailing round brackets containing metadata keywords - END
(?i)\s*\([^)]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d|\bupdated\b|\bdated\b)[^)]*\)\s*$

# General: trailing square brackets containing metadata keywords - END
(?i)\s*\[[^\]]*(?:\bversion\b|\bamend|\boppose|\bsponsor|\bsource\b|\bcommittee\b|\bletter\b|\bbill\b|\bintroduced\b|\bconcept\b|\bprior\b|\bverified|\breceived\b|\bsubmitted\b|\breflects|\bunable\b|\bper\s|\bas\sof\b|\bnone\s|\bsection\s*\d|\bupdated\b|\bdated\b)[^\]]*\]\s*$

# Trailing "Hearing Date:" metadata (strips from "Hearing Date:" onward) - END
(?i),?\s*Hearing Date:.*$

# Trailing "GOVERNOR'S VETO MESSAGE" metadata (strips the phrase + everything after) - END
(?i)\s*:?\s*GOVERNOR.?S\s+VETO\s+MESSAGE.*$

# Trailing bare date stamps with year component - END
\s+-\s*\d{1,2}/\d{1,2}/\d{2,4}\s*$
(?<=\D)\s+\d{1,2}/\d{1,2}/\d{2,4}\s*$

# Trailing "None Received" - END
(?i)\s+None Received\s*$

# Trailing bare position words - END
(?i),?\s+oppose\s*$
(?i)\s+OPPOSED\s*$
(?i),?\s+Continued\s*$
(?i),\s+co-?sponsors?\s*$
(?i),\s+sponsors?\s*$
(?i),\s+supports?\s*$

# Trailing sponsor tag: any delimiter (space/comma/dash/slash/quote) + optional narrative
# phrasing + optional "co-" prefix. Consumes the whole (co-)?sponsor token so an internal
# "co-sponsor" hyphen is never mistaken for the delimiter. - END
# (verified: no legitimate org name ends in "sponsor"/"co-sponsor" - all bill-relationship metadata)
(?i)[\s,/"-]+(?:principal\s+)?(?:is\s+|also\s+)?(?:(?:a|the|one\s+of\s+the)\s+)?(?:bill'?s?\s+)?(?:co[\s-]?)?sponsors?\s*$

# Trailing "None on file" suffix - END
(?i),?\s*None on file\s*$

# Square brackets - END
(?i)\s*\[co-sponsor\]\s*$
(?i)\s*\[sponsor\]\s*$
(?i)\s*\[sponsors\]\s*$
(?i)\s*\[co-source\]\s*$
(?i)\s*\[prior version\]\s*$
(?i)\s*\[oppose\]\s*$
(?i)\s*\[oppose unless amended\]\s*$
(?i)\s*\["in concept"\]\s*$
(?i)\s*\[principal co-sponsor\]\s*$
(?i)\s*\[co-sponsors\]\s*$
(?i)\s*\[previous version\]\s*$
(?i)\s*\[as of prior version\]\s*$
(?i)\s*\[unable to re-verify at time of writing\]\s*$

(?i)\s*\[prior to latest amendments\]\s*$
(?i)\s*\[per\s[^\]]*\]\s*$
(?i)\s*\[as of\s[^\]]*\]\s*$

# Trailing "(if)", "(as)", "(c osponsor )", "(Recommendation)", "(page)", "(Member)" metadata - END

# Trailing "(UNREG)", "(continued)", "(cont.)", "(Fall)", "(pending)" metadata flags - END
(?i)\s*\(UNREG\)\s*$
(?i)\s*\(continued\)\s*$
(?i)\s*\(cont\.\)\s*$
(?i)\s*\(Fall\)\s*$
(?i)\s*\(pending\)\s*$

# Trailing stance/procedural annotations in parens - END
(?i)\s*\(concerns?\)\s*$
(?i)\s*\(neutral\)\s*$
(?i)\s*\(watch\)\s*$
(?i)\s*\(with\)\s*$
(?i)\s*\(in\)\s*$
(?i)\s*\(Unless[^)]*\)\s*$
(?i)\s*\(Judicial discretion[^)]*\)\s*$

# Trailing "--" or "-" stance/procedural annotations - END
(?i)\s*-{1,2}\s*Neutral\s*$
(?i)\s*-{1,2}\s*END\s*$
(?i)\s*-{1,2}\s*Oppose\s*$
(?i)\s*-\s*Verified\b.*$

# Trailing "also" metadata - END
(?i)\s+also\s*$

# Trailing bare date in parens like (5/22/98) or (6/14/07) - END
\s*\(\d{1,2}/\d{1,2}/\d{2,4}\)\s*$

# Trailing "(N letters)" or "(N individuals)" metadata suffix - END
(?i)\s*\(\d+\s+(?:letters|individuals|members)\)\s*$
(?i)\s*-\s*\(\d+\s+(?:letters|individuals|members)\)\s*$
(?i),\s*\d+\s+(?:personal\s+)?letters\s*$

# Trailing count-tally "N letters/individuals/members/signatures/residents" in other delimiters (dash / bracket / paren / colon / comma / bare) - END
# Comma/bare forms require a space after the number's preceding delimiter so thousands separators ("22,743") and org local-numbers ("Local 721") are never split.
(?i)\s*[-–—]\s*(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:letters?|individuals?|members?|signatures?|residents?)\s*$
(?i)\s*\[\s*(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:letters?|individuals?|members?|signatures?|residents?)\s*\]\s*$
(?i)\s*\(\s*(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:letters?|individuals?|members?|signatures?|residents?)\s*\)\s*$
(?i)\s*:\s*(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:letters?|individuals?|members?|signatures?|residents?)\s*$
(?i),\s+(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:individuals?|signatures?|residents?)\s*$
(?i)\s+(?:approximately\s+|about\s+|over\s+|more\s+than\s+|nearly\s+)?\d[\d,]*\s+(?:individuals?|signatures?)\s*$

# Trailing "None to date" suffix (with or without space) - END
(?i)(?<=\S)[,\s]*None to date\s*$

# Trailing "- Oppose Unless" or similar stance annotations with dash - END
(?i)\s*-\s*Oppose Unless.*$

# Trailing procedural parens: (press release), ("approve"), (co-ssource), (co-source), etc. - END
(?i)\s*\(press release\)\s*$
(?i)\s*\("approve"\)\s*$
(?i)\s*\(co-?s?source\)\s*$
(?i)\s*\(\d+\s+property owners?\)\s*$
(?i)\s*\(club membership[^)]*\)\s*$
(?i)\s*\(as to[^)]*\)\s*$
(?i)\s*\(which consists[^)]*\)\s*$
(?i)\s*\(re:[^)]*\)\s*$

# Trailing ", this bill" or ", this" fragment - END
(?i),?\s*this bill\s*$
(?i),\s*this\s*$

# Trailing bare "0" artifact - END
(?i)\s+0\s*$

# Trailing date in square brackets like [May 22, 2012] - END
(?i)\s*\[(?:January|February|March|April|May|June|July|August|September|October|November|December)\s+\d{1,2},?\s*\d{4}\]\s*$

# Trailing "Unknown" suffix (with or without space/comma, requires preceding content) - END
(?i)(?<=\S)[,\s]*Unknown\s*$

# Trailing procedural annotations in parens - END
(?i)\s*\([^)]*see attach[^)]*\)\s*$

(?i)\s*\(if amended\)\s*$

# Trailing ", if" fragment - END

# Trailing enrollment/vote numbers after school district keywords (e.g. "ALAMEDA CITY UNIFIED 10,295") - END

# Bill-position / legislative prefixes (START) - approved Q26
# Strip legislative-analysis boilerplate prepended onto a real org name.
# Case-sensitive as written (do NOT add (?i)); all anchored to START.
# Verified: 0 false positives against near-misses (Page and Turnbull, Page Avenue
# Mutual Water Company, Sponsors for Educational Opportunity, In-Home Supportive
# Services, SBC, Page Mill Partners). Do NOT add a bare ^Page or ^SB strip.
# NOTE: load_cleaning_patterns() strips each line, so a literal trailing space
# in a regex is lost. The single-letter page header MUST keep an explicit
# trailing-whitespace matcher (\s) or "^Page [A-Z]" would over-strip real orgs
# like "Page Mill Partners" / "Page Avenue Mutual Water Company".
^Page [A-Z]\s
^SB \d+, Page\s
^Sponsor(?: (?:of|and supporter of) the bill)?[\s,:\-]+(?:the )?
^In support of the bill,? (?:the )?
^OPPOSITION \(UNLESS AMENDED\)
